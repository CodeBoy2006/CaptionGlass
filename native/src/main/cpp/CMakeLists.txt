cmake_minimum_required(VERSION 3.22.1)
project(captionglass LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES
        LLAMA_BUILD_SERVER LLAMA_BUILD_APP GGML_NATIVE GGML_BACKEND_DL GGML_CPU_KLEIDIAI GGML_OPENMP)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
get_filename_component(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../artifacts/deps" ABSOLUTE)
# A static ICD loader discovers the optional system driver only when OpenCL is selected.
set(OPENCL_ICD_LOADER_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENCL_LAYERS OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENCL_LAYERINFO OFF CACHE BOOL "" FORCE)
set(OPENCL_ICD_LOADER_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENCL_ICD_LOADER_HEADERS_DIR "${DEPS}/opencl-headers" CACHE PATH "" FORCE)
add_subdirectory("${DEPS}/opencl-loader" opencl-loader)
set(OpenCL_INCLUDE_DIR "${DEPS}/opencl-headers" CACHE PATH "" FORCE)
set(OpenCL_LIBRARY OpenCL CACHE STRING "" FORCE)
set(GGML_OPENCL ON CACHE BOOL "" FORCE)
set(Vulkan_INCLUDE_DIR "${DEPS}/vulkan-headers/include" CACHE PATH "" FORCE)
set(Vulkan_GLSLC_EXECUTABLE "${DEPS}/shaderc-build/glslc/glslc" CACHE FILEPATH "" FORCE)
set(SPIRV-Headers_DIR "${DEPS}/spirv-headers-install/share/cmake/SPIRV-Headers")
find_package(SPIRV-Headers CONFIG REQUIRED)
find_program(VULKAN_HOST_CC clang PATHS /usr/bin /usr/local/bin REQUIRED NO_DEFAULT_PATH)
find_program(VULKAN_HOST_CXX clang++ PATHS /usr/bin /usr/local/bin REQUIRED NO_DEFAULT_PATH)
configure_file(vulkan-host.cmake.in vulkan-host.cmake @ONLY)
set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${CMAKE_CURRENT_BINARY_DIR}/vulkan-host.cmake" CACHE FILEPATH "" FORCE)
add_subdirectory(../../../../artifacts/deps/llama-source llama-build)
# Register accelerators only when selected. CPU mode must not initialize device drivers.
get_target_property(GGML_DEFINITIONS ggml COMPILE_DEFINITIONS)
list(REMOVE_ITEM GGML_DEFINITIONS GGML_USE_VULKAN GGML_USE_OPENCL)
set_property(TARGET ggml PROPERTY COMPILE_DEFINITIONS "${GGML_DEFINITIONS}")
set(HEXAGON "${DEPS}/hexagon")
if(NOT EXISTS "${HEXAGON}/verified")
    message(FATAL_ERROR "Run bash scripts/prepare-hexagon.sh before building (requires Docker).")
endif()
set(HEXAGON_SOURCE "${DEPS}/llama-source/ggml/src/ggml-hexagon")
add_library(captionglass-hexagon STATIC
    "${HEXAGON_SOURCE}/ggml-hexagon.cpp" "${HEXAGON_SOURCE}/htp-drv.cpp"
    "${HEXAGON}/generated/htp_iface_stub.c")
target_include_directories(captionglass-hexagon PRIVATE
    "${DEPS}/llama-source/ggml/src" "${HEXAGON_SOURCE}" "${HEXAGON_SOURCE}/htp"
    "${HEXAGON}/generated" "${HEXAGON}/include/incs" "${HEXAGON}/include/incs/stddef")
target_link_libraries(captionglass-hexagon PRIVATE ggml-base log dl)
target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
add_library(captionglass SHARED translator.cpp resample.cpp "${DEPS}/sherpa-source/sherpa-onnx/csrc/resample.cc")
target_include_directories(captionglass PRIVATE "${DEPS}/sherpa-source")
target_link_libraries(captionglass PRIVATE llama captionglass-hexagon log)
target_link_options(captionglass PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,libOpenCL.a")
