cmake_minimum_required(VERSION 3.22.1)
project(captionglass LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES
        LLAMA_BUILD_SERVER LLAMA_BUILD_APP GGML_NATIVE GGML_BACKEND_DL GGML_CPU_KLEIDIAI GGML_OPENMP)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
get_filename_component(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../artifacts/deps" ABSOLUTE)
set(Vulkan_INCLUDE_DIR "${DEPS}/vulkan-headers/include" CACHE PATH "" FORCE)
set(Vulkan_GLSLC_EXECUTABLE "${DEPS}/shaderc-build/glslc/glslc" CACHE FILEPATH "" FORCE)
set(SPIRV-Headers_DIR "${DEPS}/spirv-headers-install/share/cmake/SPIRV-Headers")
find_package(SPIRV-Headers CONFIG REQUIRED)
find_program(VULKAN_HOST_CC clang PATHS /usr/bin /usr/local/bin REQUIRED NO_DEFAULT_PATH)
find_program(VULKAN_HOST_CXX clang++ PATHS /usr/bin /usr/local/bin REQUIRED NO_DEFAULT_PATH)
configure_file(vulkan-host.cmake.in vulkan-host.cmake @ONLY)
set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${CMAKE_CURRENT_BINARY_DIR}/vulkan-host.cmake" CACHE FILEPATH "" FORCE)
add_subdirectory(../../../../artifacts/deps/llama-source llama-build)
target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
add_library(captionglass SHARED translator.cpp)
target_link_libraries(captionglass PRIVATE llama log)
target_link_options(captionglass PRIVATE "-Wl,-z,max-page-size=16384")
